# This repository contains code to reproduce results from the paper: *A molecular map of the living human brain from quantitative MRI*

DLPFC = dorsolateral prefrontal cortex;
GO = gene ontolgy;
LOOCV = leave-one-out cross validation;
PD = Parkinson's disease;
qMRI = quantitative magnetic resonance imaging;
SRT = spatially-resolved transcriptomics
